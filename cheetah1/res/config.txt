#Cheetah Engine configuration file
RESOURCE_PACK pack=auschwitz
TEXTURE_FILTER filter=Nearest
GAME_GRAPHICS graphics=High
CLEAR_LIGHTS lights=false
MIPMAP_LEVEL mipMapping=0.4
ANISOTROPIC_LEVEL anisotropic=4
POP_IN popin=15
PARTICLES_POP_IN pPopin=5
GRASS_POP_IN gPopin=10
LIGHT_POP_IN lightPopin=10
GRAVITY gravity=9.8
PARTICLES_LEVEL particles=2
EFFECTS_AUDIO_LEVEL effects=20
GOD=false
#Skips drawing an object under a point/spot light pass once it's outside
#that light's range. Off by default until verified visually - flip to
#true once torches/lampposts look right with it on.
LIGHT_RANGE_CULLING culling=false
#Skips rendering entities entirely outside the camera's view frustum.
#Off by default until verified visually - flip to true once nothing
#pops in/out incorrectly at the edges of the screen.
FRUSTUM_CULLING culling=false
#Distance tiers for enemy AI/collision update frequency: closer than
#midDist ticks every frame, midDist..farDist every 2nd tick, beyond
#farDist every farTickSkip-th tick. Not geometric LOD (these are
#billboard sprites, nothing to simplify) - just how often distant
#enemies think, to save CPU on checkCollisions/line-of-sight calls.
LOD_MID_DISTANCE midDist=10
LOD_FAR_DISTANCE farDist=20
LOD_FAR_TICK_SKIP farTickSkip=4
#Size of the pooled OpenAL source pool used for one-shot 3D sounds
#(weapon fire, pickups, impacts). When every source is busy, the
#least-recently-used one is recycled rather than growing the pool.
MAX_AUDIO_SOURCES sources=16
#Display mode read once at startup (Main.java) before the window is
#created. Changing these from the in-game settings menu requires a
#restart to take effect - there's no live resize path.
TARGET_WIDTH width=1280
TARGET_HEIGHT height=720
FULLSCREEN fullscreen=true
